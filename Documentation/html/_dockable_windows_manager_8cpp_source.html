32void DockableWindowsManager::begin() {
33 // Make the dockspace window cover the entire usable viewport area.
34 ImGuiViewport* viewport = ImGui::GetMainViewport();
35 ImGui::SetNextWindowPos(viewport->WorkPos);
36 ImGui::SetNextWindowSize(viewport->WorkSize);
37 ImGui::SetNextWindowViewport(viewport->ID);
38
39 // Flags: no decorations, just a container for the docking area.
40 ImGuiWindowFlags dockspace_flags =
41 ImGuiWindowFlags_NoDocking |
42 ImGuiWindowFlags_NoTitleBar |
43 ImGuiWindowFlags_NoCollapse |
44 ImGuiWindowFlags_NoResize |
45 ImGuiWindowFlags_NoMove |
46 ImGuiWindowFlags_NoBringToFrontOnFocus |
47 ImGuiWindowFlags_NoNavFocus;
48
49 // Remove padding and borders so the dockspace fills the window precisely.
50 ImGui::PushStyleVar(ImGuiStyleVar_WindowRounding, 0.0f);
51 ImGui::PushStyleVar(ImGuiStyleVar_WindowBorderSize, 0.0f);
52 ImGui::PushStyleVar(ImGuiStyleVar_WindowPadding, ImVec2(0.0f, 0.0f));
53 ImGui::Begin("DockSpace", nullptr, dockspace_flags);
54 ImGui::PopStyleVar(3);
55
56 // Create the actual dockspace node.
57 ImGuiID dockspace_id = ImGui::GetID("MainDockSpace");
58 ImGui::DockSpace(dockspace_id);
59
60 // On the very first frame, decide whether to restore saved layout or apply current.
61 if (m_first_frame) {
62 m_first_frame = false;
63 if (!hasIniFile() && m_has_layout) {
64 m_layout_valid = false;
65 }
66 }
67
68 // If the layout is marked invalid and we have a layout + windows, rebuild now.
69 if (!m_layout_valid && m_has_layout && !m_windows.empty()) {
70 applyLayout(dockspace_id, m_current_layout);
71 m_layout_valid = true;
72 }
73}

90void DockableWindowsManager::applyLayout(ImGuiID dockspace_id, const DockLayoutNode& layout) {
91 // Wipe any existing layout and create a fresh dockspace node.
92 ImGui::DockBuilderRemoveNode(dockspace_id);
93 ImGui::DockBuilderAddNode(dockspace_id, ImGuiDockNodeFlags_DockSpace);
94 ImGui::DockBuilderSetNodeSize(dockspace_id, ImGui::GetMainViewport()->WorkSize);
95
96 // Recursively build the node tree.
97 std::function<void(ImGuiID, const DockLayoutNode&)> build_node;
98 build_node = [&](ImGuiID node_id, const DockLayoutNode& node) {
99 if (node.type == DockLayoutNode::Window) {
100 // Leaf: dock a named window into this node.
101 ImGui::DockBuilderDockWindow(node.window_name.c_str(), node_id);
102 }
103 else if (node.type == DockLayoutNode::Split) {
104 // Split the node into two children and recurse.
105 ImGuiID child1_id, child2_id;
106 ImGui::DockBuilderSplitNode(node_id, node.split_dir, node.split_ratio,
107 &child1_id, &child2_id);
108 build_node(child1_id, *node.child1);
109 build_node(child2_id, *node.child2);
110 }
111 };
112 build_node(dockspace_id, layout);
113
114 // Finalize the docking layout.
115 ImGui::DockBuilderFinish(dockspace_id);
116}

118bool DockableWindowsManager::hasIniFile() const {
119 const char* ini_file = ImGui::GetIO().IniFilename;
120 if (!ini_file || !*ini_file) return false;
121 std::ifstream file(ini_file, std::ios::binary | std::ios::ate);
122 // File exists and has more than a minimal size -> saved layout present.
123 return file.good() && file.tellg() > 2;
