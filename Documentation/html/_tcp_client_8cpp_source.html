42bool TcpClient::connectToServer() {
43 // Socket creation with IPv4 address family
44#ifdef _WIN32
45 m_socket = socket(AF_INET, SOCK_STREAM, 0);
46 if (m_socket == INVALID_SOCKET) return false;
47#else
48 m_socket = socket(AF_INET, SOCK_STREAM, 0); // IpV4; stream socket; TCP.
49 if (m_socket < 0) return false; // No sockets left or incorrect params were passed to function.
50#endif
51
52 struct sockaddr_in addr; // Server address structure.
53 addr.sin_family = AF_INET; // IPv4.
54 addr.sin_port = htons(m_port); // Port (Host TO Network Short, little endian to big endian).
55 inet_pton(AF_INET, m_server_address.c_str(), &addr.sin_addr); // Ip to bytes.
56
57 // Tcp connection. 3-way handshake.
58 if (connect(m_socket, (struct sockaddr*)&addr, sizeof(addr)) < 0) {
59#ifdef _WIN32
60 closesocket(m_socket);
61#else
62 close(m_socket);
63#endif
64 m_socket = -1;
65 return false;
66 }
67
68 m_is_server_connected = true;
69
70 setTimeouts(5000, 5000);
71
72 if (m_use_tls) {
73 m_ssl = SSL_new(m_ssl_ctx);
74 SSL_set_fd(m_ssl, m_socket);
75 if (SSL_connect(m_ssl) <= 0) {
76 ERR_print_errors_fp(stderr);
77 cleanupSsl();
78 closeSocket();
79 return false;
80 }
81 }
82
83 return true;
84}

86std::string TcpClient::sendRequest(const std::string& request_json) {
87 // Return an error immediately if the client is not connected.
88 if (!m_is_server_connected)
89 return R"({"status":"error","message":"Not connected"})";
90
91 // Append a newline character to mark the end of the request.
92 std::string data = request_json + "\n";
93
94 // Send the request data.
95 // For TLS connections we use SSL_write(); otherwise we use plain send().
96 int total_sent = 0;
97 int len = static_cast<int>(data.size());
98 while (total_sent < len) {
99 int sent = 0;
100 if (m_use_tls) {
101 // SSL_write returns the number of bytes written, or <= 0 on error.
102 sent = SSL_write(m_ssl, data.c_str() + total_sent, len - total_sent);
103 }
104 else {
105 // On Windows MSG_NOSIGNAL is not defined; use 0 instead.
106 // On Linux MSG_NOSIGNAL prevents the SIGPIPE signal if the socket is broken.
107#ifdef _WIN32
108 sent = send(m_socket, data.c_str() + total_sent, len - total_sent, 0);
109#else
110 sent = send(m_socket, data.c_str() + total_sent, len - total_sent, MSG_NOSIGNAL);
111#endif
112 }
113
114 // If send fails, disconnect and return an error response.
115 if (sent <= 0) {
116 disconnect();
117 return R"({"status":"error","message":"Send failed, disconnected"})";
118 }
119 total_sent += sent;
120 }
121
122 // Receive the response.
123 // For TLS we use SSL_read(); otherwise we use plain recv().
124 std::string response;
125 char buf[4096];
126 while (true) {
127 memset(buf, 0, sizeof(buf));
128 int bytes_read = 0;
129 if (m_use_tls) {
130 // SSL_read returns the number of bytes read, or <= 0 on error/EOF.
131 bytes_read = SSL_read(m_ssl, buf, sizeof(buf) - 1);
132 }
133 else {
134#ifdef _WIN32
135 bytes_read = recv(m_socket, buf, sizeof(buf) - 1, 0);
136#else
137 bytes_read = recv(m_socket, buf, sizeof(buf) - 1, 0);
138#endif
139 }
140
141 if (bytes_read <= 0) {
142 // Connection closed or error occurred.
143 m_is_server_connected = false;
144 closeSocket();
145 return R"({"status":"error","message":"Server disconnected"})";
146 }
147
148 // Append received data to the leftover buffer.
149 m_leftover_buffer.append(buf, bytes_read);
150
151 // If a complete line (terminated by '\n') is found, stop reading.
152 if (m_leftover_buffer.find('\n') != std::string::npos) {
153 break;
154 }
155 }
156
157 // Extract the first complete JSON line from the buffer.
158 size_t newline_pos = m_leftover_buffer.find('\n');
159 if (newline_pos == std::string::npos) {
160 // This should not happen because we already checked for '\n'.
161 return R"({"status":"error","message":"Invalid server response"})";
162 }
163
164 response = m_leftover_buffer.substr(0, newline_pos);
165
166 // Remove the used line (including the newline) from the buffer,
167 // leaving any remaining data for the next read.
168 m_leftover_buffer.erase(0, newline_pos + 1);
169
170 return response;
171}

178void TcpClient::setTimeouts(int send_timeout_ms, int recv_timeout_ms) {
179 if (m_socket != -1) {
180#ifdef _WIN32
181 DWORD timeout = send_timeout_ms;
182 setsockopt(m_socket, SOL_SOCKET, SO_SNDTIMEO, (const char*)&timeout, sizeof(timeout));
183 timeout = recv_timeout_ms;
184 setsockopt(m_socket, SOL_SOCKET, SO_RCVTIMEO, (const char*)&timeout, sizeof(timeout));
185#else
186 struct timeval tv;
187 tv.tv_sec = send_timeout_ms / 1000;
188 tv.tv_usec = (send_timeout_ms % 1000) * 1000;
189 setsockopt(m_socket, SOL_SOCKET, SO_SNDTIMEO, &tv, sizeof(tv));
190 tv.tv_sec = recv_timeout_ms / 1000;
191 tv.tv_usec = (recv_timeout_ms % 1000) * 1000;
192 setsockopt(m_socket, SOL_SOCKET, SO_RCVTIMEO, &tv, sizeof(tv));
193#endif
194 m_has_timeouts = true;
195 }
196}
