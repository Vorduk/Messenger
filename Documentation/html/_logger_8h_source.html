91 fmt::format_string<Args...> fmt, Args&&... args) {

96 if (m_show_file_info && m_show_level_prefix) {

98 m_logger->log(lvl, "[{}:{} {}] {}",

101 else if (m_show_file_info) {

104 extractFileName(file), line, func, message);
105 }
106 else if (m_show_level_prefix) {
107 // Only level prefix.
108 m_logger->log(lvl, message);
109 }
110 else {
111 // Without prefix.
112 m_logger->log(lvl, message);
113 }
114 }

116 // Extracts the base filename from a full path.
117 static const char* extractFileName(const char* path) {
118 const char* lastSlash = strrchr(path, '/');
119 const char* lastBackslash = strrchr(path, '\\');
120 const char* filename = path;
121
122 if (lastSlash) filename = lastSlash + 1;
123 if (lastBackslash && lastBackslash > filename) filename = lastBackslash + 1;
124
125 return filename;
126 }
