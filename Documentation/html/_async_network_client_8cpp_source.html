80void AsyncNetworkClient::workerLoop() {

82 std::unique_lock<std::mutex> lock(m_mutex); // Acquire the mutex to check the queue safely.
83 m_cv.wait(lock, [this] { return !m_queue.empty() || !m_is_running; }); // Sleep until a request arrives or stop is requested.
84 if (!m_is_running) break; // Exit the loop if destruction has been requested.

87 m_queue.pop();

95void AsyncNetworkClient::reconnectLoop() {
96 while (m_is_running && m_auto_reconnect) {
97 std::unique_lock<std::mutex> lock(m_reconnect_mutex);
98 m_reconnect_cv.wait_for(lock, std::chrono::milliseconds(m_reconnect_interval_ms),
99 [this] { return !m_is_running || !m_should_reconnect; }); // Wait for reconnection interval or notification.
100
101 if (!m_is_running) break; // Exit the loop if destruction has been requested.
102
103 // Only attempt reconnection if we are disconnected
104 if (!m_is_server_connected && m_should_reconnect && !m_reconnect_in_progress) {
105 m_reconnect_in_progress = true; // Mark reconnection as in progress.
106 lock.unlock();
107

111 m_should_reconnect = false;

114 // Notify the application via the callback on the main thread
